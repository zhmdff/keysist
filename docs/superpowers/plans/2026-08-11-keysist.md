# Keysist Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Windows tray app that fires app-launch or key-remap actions from specific *physical* keyboard/HID devices, per-key by default, with a GUI to discover devices and edit mappings.

**Architecture:** A hidden Raw Input message window identifies which device sent a keystroke; a low-level keyboard hook (`WH_KEYBOARD_LL`) is correlated against it (via a short-TTL pending-event queue) to optionally swallow the keystroke; a config store maps `(devicePath, key)` to an action (launch process or send synthetic keys); a WPF/WinForms tray app hosts a Discover Device window and an Edit Mappings window that write the config.

**Tech Stack:** C# / .NET 8 (`net8.0-windows`), WPF (editor windows) + WinForms `NotifyIcon` (tray icon), `System.Text.Json` for config, xUnit for pure-logic unit tests, raw Win32 P/Invoke for Raw Input / `WH_KEYBOARD_LL` / `SendInput` (no external interop package — declare only what's needed).

## Global Constraints

- Target framework: `net8.0-windows` for all projects (Raw Input / hooks / WPF are Windows-only).
- Config file location: `%AppData%\Keysist\config.json` (from spec).
- Default `swallow` behavior: eat the keystroke unless a mapping explicitly sets `swallow: false` (from spec).
- Never eat a keystroke without a confirmed Raw-Input↔hook correlation (from spec) — unmatched events always pass through.
- Config parse failure must keep the last-good in-memory config and never crash the app (from spec).
- No external NuGet packages beyond `Microsoft.NET.Test.Sdk` + `xunit` + `xunit.runner.visualstudio` for the test project. Everything else is BCL/WinForms/WPF/P-Invoke — keep dependencies minimal (YAGNI).

```csharp
namespace Keysist.Config;

public static class DeviceMatcher
{
    public static DeviceConfig? FindMatch(IEnumerable<DeviceConfig> devices, string actualDevicePath)
    {
        foreach (var device in devices)
        {
            if (!string.IsNullOrEmpty(device.DevicePathMatch) &&
                actualDevicePath.Contains(device.DevicePathMatch, StringComparison.OrdinalIgnoreCase))
            {
                return device;
            }
        }
        return null;
    }
}
```

```csharp
// src/Keysist/Config/ConfigStore.cs
using System.Text.Json;

namespace Keysist.Config;

public class ConfigStore
{
    private readonly string _filePath;
    private readonly JsonSerializerOptions _jsonOptions = new() { PropertyNameCaseInsensitive = true };
    private FileSystemWatcher? _watcher;
    private System.Threading.Timer? _debounceTimer;

    public KeysistConfig Current { get; private set; } = new();
    public event Action<string>? LoadFailed;

    public ConfigStore(string filePath)
    {
        _filePath = filePath;
    }

    public void Load()
    {
        if (!File.Exists(_filePath))
        {
            Current = new KeysistConfig();
            return;
        }

        try
        {
            var json = File.ReadAllText(_filePath);
            var parsed = JsonSerializer.Deserialize<KeysistConfig>(json, _jsonOptions);
            if (parsed is null) throw new JsonException("Deserialized to null");
            Current = parsed;
        }
        catch (Exception ex) when (ex is JsonException or IOException)
        {
            LoadFailed?.Invoke(ex.Message);
        }
    }

    public void StartWatching()
    {
        var dir = Path.GetDirectoryName(_filePath)!;
        Directory.CreateDirectory(dir);
        _watcher = new FileSystemWatcher(dir, Path.GetFileName(_filePath))
        {
            EnableRaisingEvents = true
        };
        _watcher.Changed += (_, _) => DebouncedLoad();
        _watcher.Created += (_, _) => DebouncedLoad();
    }

    public void StopWatching()
    {
        _watcher?.Dispose();
        _watcher = null;
        _debounceTimer?.Dispose();
        _debounceTimer = null;
    }

    private void DebouncedLoad()
    {
        _debounceTimer?.Dispose();
        _debounceTimer = new System.Threading.Timer(_ => Load(), null, 300, System.Threading.Timeout.Infinite);
    }
}
```

```csharp
// src/Keysist/Correlation/InputCorrelator.cs
namespace Keysist.Correlation;

public class InputCorrelator
{
    private readonly IClock _clock;
    private readonly long _ttlMs;
    private readonly Dictionary<int, Queue<(string DevicePath, long RecordedAtMs)>> _pending = new();
    private readonly object _lock = new();

    public InputCorrelator(IClock clock, int ttlMs = 15)
    {
        _clock = clock;
        _ttlMs = ttlMs;
    }

    public void RecordRawInput(int vkey, string devicePath)
    {
        lock (_lock)
        {
            if (!_pending.TryGetValue(vkey, out var queue))
            {
                queue = new Queue<(string, long)>();
                _pending[vkey] = queue;
            }
            queue.Enqueue((devicePath, _clock.NowMs()));
        }
    }

    public bool TryCorrelate(int vkey, out string? devicePath)
    {
        lock (_lock)
        {
            devicePath = null;
            if (!_pending.TryGetValue(vkey, out var queue))
                return false;

            var now = _clock.NowMs();
            while (queue.Count > 0)
            {
                var (path, recordedAt) = queue.Dequeue();
                if (now - recordedAt <= _ttlMs)
                {
                    devicePath = path;
                    return true;
                }
            }
            return false;
        }
    }
}
```

```csharp
// src/Keysist/Actions/SendInputKeySender.cs
using System.Runtime.InteropServices;
using Keysist.Config;

namespace Keysist.Actions;

public class SendInputKeySender : IKeySender
{
    [StructLayout(LayoutKind.Sequential)]
    private struct INPUT
    {
        public uint Type;
        public KEYBDINPUT Ki;
        public ulong Padding; // pad union to MOUSEINPUT size
    }

    [StructLayout(LayoutKind.Sequential)]
    private struct KEYBDINPUT
    {
        public ushort VirtualKeyCode;
        public ushort ScanCode;
        public uint Flags;
        public uint Time;
        public IntPtr ExtraInfo;
    }

    private const uint InputTypeKeyboard = 1;
    private const uint KeyEventFKeyUp = 0x0002;
    private const uint KeyEventFUnicode = 0x0004;

    [DllImport("user32.dll", SetLastError = true)]
    private static extern uint SendInput(uint numInputs, INPUT[] inputs, int sizeOfInputStruct);

    public void Send(IReadOnlyList<SendStep> steps)
    {
        foreach (var step in steps)
        {
            if (step.DelayMs > 0) Thread.Sleep(step.DelayMs);

            if (step.Send == "vkey")
            {
                var vk = VirtualKeyNames.Resolve(step.Value);
                SendKeyDownUp(vk);
            }
            else if (step.Send == "text")
            {
                foreach (var ch in step.Value)
                    SendUnicodeChar(ch);
            }
        }
    }

    private void SendKeyDownUp(ushort vk)
    {
        var down = new INPUT { Type = InputTypeKeyboard, Ki = new KEYBDINPUT { VirtualKeyCode = vk } };
        var up = new INPUT { Type = InputTypeKeyboard, Ki = new KEYBDINPUT { VirtualKeyCode = vk, Flags = KeyEventFKeyUp } };
        SendInput(2, new[] { down, up }, Marshal.SizeOf<INPUT>());
    }

    private void SendUnicodeChar(char ch)
    {
        var down = new INPUT { Type = InputTypeKeyboard, Ki = new KEYBDINPUT { ScanCode = ch, Flags = KeyEventFUnicode } };
        var up = new INPUT { Type = InputTypeKeyboard, Ki = new KEYBDINPUT { ScanCode = ch, Flags = KeyEventFUnicode | KeyEventFKeyUp } };
        SendInput(2, new[] { down, up }, Marshal.SizeOf<INPUT>());
    }
}
```

```csharp
// src/Keysist/Actions/VirtualKeyNames.cs
namespace Keysist.Actions;

public static class VirtualKeyNames
{
    private static readonly Dictionary<string, ushort> ByName = new(StringComparer.OrdinalIgnoreCase)
    {
        ["VK_MEDIA_PLAY_PAUSE"] = 0xB3,
        ["VK_MEDIA_NEXT_TRACK"] = 0xB0,
        ["VK_MEDIA_PREV_TRACK"] = 0xB1,
        ["VK_VOLUME_UP"] = 0xAF,
        ["VK_VOLUME_DOWN"] = 0xAE,
        ["VK_VOLUME_MUTE"] = 0xAD,
    };

    public static ushort Resolve(string name)
    {
        if (ByName.TryGetValue(name, out var vk)) return vk;
        if (name.StartsWith("0x") && ushort.TryParse(name[2..], System.Globalization.NumberStyles.HexNumber, null, out var hex))
            return hex;
        throw new ArgumentException($"Unknown virtual key name '{name}'");
    }
}
```

```csharp
public static class DeviceIdentity
{
    public static string ResolveDevicePath(IntPtr hDevice)
    {
        uint size = 0;
        RawInputNative.GetRawInputDeviceInfo(hDevice, RawInputNative.RIDI_DEVICENAME, IntPtr.Zero, ref size);
        if (size == 0) return "";

        var buffer = Marshal.AllocHGlobal((int)size * 2); // wide chars
        try
        {
            var written = RawInputNative.GetRawInputDeviceInfo(hDevice, RawInputNative.RIDI_DEVICENAME, buffer, ref size);
            if (written <= 0) return "";
            return Marshal.PtrToStringUni(buffer) ?? "";
        }
        finally
        {
            Marshal.FreeHGlobal(buffer);
        }
    }
```

```csharp
using System.Runtime.InteropServices;
using Keysist.RawInputInterop;

namespace Keysist.RawInputInterop;

public class RawInputWindow : System.Windows.Forms.NativeWindow, IDisposable
{
    public event Action<int Vkey, string DevicePath>? RawKeyDown;

    public void Start()
    {
        CreateHandle(new System.Windows.Forms.CreateParams
        {
            Caption = "KeysistRawInput",
            Parent = new IntPtr(-3) // HWND_MESSAGE
        });

        var devices = new[]
        {
            new RawInputNative.RAWINPUTDEVICE
            {
                UsagePage = RawInputNative.HID_USAGE_PAGE_GENERIC,
                Usage = RawInputNative.HID_USAGE_GENERIC_KEYBOARD,
                Flags = RawInputNative.RIDEV_INPUTSINK,
                Target = Handle
            }
        };
        RawInputNative.RegisterRawInputDevices(devices, (uint)devices.Length, (uint)Marshal.SizeOf<RawInputNative.RAWINPUTDEVICE>());
    }

    protected override void WndProc(ref System.Windows.Forms.Message m)
    {
        if (m.Msg == RawInputNative.WM_INPUT)
        {
            HandleInput(m.LParam);
        }
        base.WndProc(ref m);
    }

    private void HandleInput(IntPtr hRawInput)
    {
        uint size = 0;
        RawInputNative.GetRawInputData(hRawInput, RawInputNative.RID_INPUT, IntPtr.Zero, ref size, (uint)Marshal.SizeOf<RawInputNative.RAWINPUTHEADER>());
        if (size == 0) return;

        var buffer = Marshal.AllocHGlobal((int)size);
        try
        {
            if (RawInputNative.GetRawInputData(hRawInput, RawInputNative.RID_INPUT, buffer, ref size, (uint)Marshal.SizeOf<RawInputNative.RAWINPUTHEADER>()) != size)
                return;

            var raw = Marshal.PtrToStructure<RawInputNative.RAWINPUT>(buffer);
            if (raw.Header.Type != RawInputNative.RIM_TYPEKEYBOARD) return;
            if ((raw.Keyboard.Flags & 1) != 0) return; // key-up (break flag) -- only fire on key-down

            var devicePath = DeviceIdentity.ResolveDevicePath(raw.Header.Device);
            RawKeyDown?.Invoke(raw.Keyboard.VKey, devicePath);
        }
        finally
        {
            Marshal.FreeHGlobal(buffer);
        }
    }

    public void Stop() => DestroyHandle();

    public void Dispose() => Stop();
}
```

```csharp
// src/Keysist/Hooking/LowLevelKeyboardHook.cs
using System.Runtime.InteropServices;
using Keysist.Correlation;

namespace Keysist.Hooking;

public class LowLevelKeyboardHook : IDisposable
{
    private const int WH_KEYBOARD_LL = 13;
    private const int WM_KEYDOWN = 0x0100;
    private const int WM_SYSKEYDOWN = 0x0104;

    [StructLayout(LayoutKind.Sequential)]
    private struct KBDLLHOOKSTRUCT
    {
        public uint VkCode;
        public uint ScanCode;
        public uint Flags;
        public uint Time;
        public IntPtr ExtraInfo;
    }

    private delegate IntPtr HookProc(int nCode, IntPtr wParam, IntPtr lParam);

    [DllImport("user32.dll", SetLastError = true)]
    private static extern IntPtr SetWindowsHookEx(int idHook, HookProc lpfn, IntPtr hMod, uint dwThreadId);

    [DllImport("user32.dll", SetLastError = true)]
    private static extern bool UnhookWindowsHookEx(IntPtr hhk);

    [DllImport("user32.dll")]
    private static extern IntPtr CallNextHookEx(IntPtr hhk, int nCode, IntPtr wParam, IntPtr lParam);

    /// <summary>Given a vkey, returns true if the key should be eaten (swallowed).</summary>
    public Func<int, bool>? ShouldEatVkey;

    private IntPtr _hookHandle = IntPtr.Zero;
    private HookProc? _hookProcDelegate; // keep alive to prevent GC of the delegate

    public void Install()
    {
        _hookProcDelegate = HookCallback;
        _hookHandle = SetWindowsHookEx(WH_KEYBOARD_LL, _hookProcDelegate, IntPtr.Zero, 0);
    }

    public void Uninstall()
    {
        if (_hookHandle != IntPtr.Zero)
        {
            UnhookWindowsHookEx(_hookHandle);
            _hookHandle = IntPtr.Zero;
        }
    }

    private IntPtr HookCallback(int nCode, IntPtr wParam, IntPtr lParam)
    {
        if (nCode >= 0 && (wParam == WM_KEYDOWN || wParam == WM_SYSKEYDOWN))
        {
            var data = Marshal.PtrToStructure<KBDLLHOOKSTRUCT>(lParam);
            if (ShouldEatVkey?.Invoke((int)data.VkCode) == true)
            {
                return (IntPtr)1; // eat
            }
        }
        return CallNextHookEx(_hookHandle, nCode, wParam, lParam);
    }

    public void Dispose() => Uninstall();
}
```

```csharp
// src/Keysist/Pipeline/InputPipeline.cs
using Keysist.Actions;
using Keysist.Config;
using Keysist.Correlation;
using Keysist.Hooking;
using Keysist.RawInputInterop;

namespace Keysist.Pipeline;

public class InputPipeline : IDisposable
{
    private readonly ConfigStore _configStore;
    private readonly RawInputWindow _rawInputWindow = new();
    private readonly LowLevelKeyboardHook _hook = new();
    private readonly InputCorrelator _correlator = new(new SystemClock(), ttlMs: 15);
    private readonly ActionDispatcher _dispatcher = new(new WindowsProcessLauncher(), new SendInputKeySender());

    public InputPipeline(ConfigStore configStore)
    {
        _configStore = configStore;
    }

    public void Start()
    {
        _rawInputWindow.RawKeyDown += OnRawKeyDown;
        _rawInputWindow.Start();

        _hook.ShouldEatVkey = OnShouldEatVkey;
        _hook.Install();
    }

    public void Stop()
    {
        _hook.Uninstall();
        _rawInputWindow.Stop();
    }

    private void OnRawKeyDown(int vkey, string devicePath)
    {
        _correlator.RecordRawInput(vkey, devicePath);
    }

    private bool OnShouldEatVkey(int vkey)
    {
        if (!_correlator.TryCorrelate(vkey, out var devicePath) || devicePath is null)
            return false; // no confirmed device -- never eat (spec: never break normal typing)

        var device = DeviceMatcher.FindMatch(_configStore.Current.Devices, devicePath);
        if (device is null) return false;

        var keyName = VirtualKeyToName(vkey);
        var (action, swallow) = ResolveAction(device, keyName);
        if (action is null) return false;

        // Dispatch off the hook thread -- hook callbacks must return fast.
        ThreadPool.QueueUserWorkItem(_ => _dispatcher.Dispatch(action));

        return swallow;
    }

    private static (ActionConfig? Action, bool Swallow) ResolveAction(DeviceConfig device, string keyName)
    {
        if (device.AnyKeyAction is not null)
            return (device.AnyKeyAction, device.DefaultSwallow);

        var mapping = device.Mappings?.FirstOrDefault(m => string.Equals(m.Key, keyName, StringComparison.OrdinalIgnoreCase));
        if (mapping is null) return (null, false);

        return (mapping.Action, mapping.Swallow ?? device.DefaultSwallow);
    }

    private static string VirtualKeyToName(int vkey) => ((System.Windows.Forms.Keys)vkey).ToString();

    public void Dispose() => Stop();
}
```

```csharp
// src/Keysist/Ui/DiscoverDeviceWindow.cs
using System.Windows.Forms;
using Keysist.Config;
using Keysist.RawInputInterop;

namespace Keysist.Ui;

public class DiscoverDeviceWindow : Form
{
    private readonly ConfigStore _configStore;
    private readonly RawInputWindow _observer = new();
    private readonly Label _status = new() { AutoSize = true, Top = 10, Left = 10, Text = "Press a key on the target device..." };
    private readonly TextBox _nameBox = new() { Top = 40, Left = 10, Width = 200, PlaceholderText = "Device name" };
    private readonly CheckBox _anyKeyBox = new() { Top = 70, Left = 10, Text = "Any key on this device (ignore key identity)" };
    private readonly TextBox _launchPathBox = new() { Top = 100, Left = 10, Width = 300, PlaceholderText = "Path to launch (leave blank to skip)" };
    private readonly Button _saveButton = new() { Top = 130, Left = 10, Text = "Save mapping" };

    private string? _lastDevicePath;
    private int? _lastVkey;

    public DiscoverDeviceWindow(ConfigStore configStore)
    {
        _configStore = configStore;
        Text = "Discover Device";
        Width = 400;
        Height = 220;

        Controls.Add(_status);
        Controls.Add(_nameBox);
        Controls.Add(_anyKeyBox);
        Controls.Add(_launchPathBox);
        Controls.Add(_saveButton);

        _saveButton.Click += (_, _) => SaveMapping();

        _observer.RawKeyDown += OnRawKeyDown;
        _observer.Start();
        FormClosed += (_, _) => _observer.Stop();
    }

    private void OnRawKeyDown(int vkey, string devicePath)
    {
        _lastVkey = vkey;
        _lastDevicePath = devicePath;
        var keyName = ((Keys)vkey).ToString();
        BeginInvoke(() => _status.Text = $"Detected: {keyName} on device path containing '{ShortPath(devicePath)}'");
    }

    private static string ShortPath(string devicePath)
    {
        var idx = devicePath.IndexOf("&PID_", StringComparison.OrdinalIgnoreCase);
        if (idx < 0) return devicePath;
        var end = devicePath.IndexOf('#', idx);
        return devicePath[..(end > 0 ? end : devicePath.Length)];
    }

    private void SaveMapping()
    {
        if (_lastDevicePath is null || _lastVkey is null)
        {
            MessageBox.Show("Press a key first.");
            return;
        }
        if (string.IsNullOrWhiteSpace(_nameBox.Text))
        {
            MessageBox.Show("Enter a device name.");
            return;
        }

        var match = ShortPath(_lastDevicePath);
        var device = _configStore.Current.Devices.FirstOrDefault(d => d.DevicePathMatch == match);
        if (device is null)
        {
            device = new DeviceConfig { Name = _nameBox.Text, DevicePathMatch = match, DefaultSwallow = true };
            _configStore.Current.Devices.Add(device);
        }

        var action = new ActionConfig
        {
            Type = "launch",
            Path = _launchPathBox.Text
        };

        if (_anyKeyBox.Checked)
        {
            device.AnyKeyAction = action;
            device.Mappings = null;
        }
        else
        {
            var keyName = ((Keys)_lastVkey.Value).ToString();
            device.Mappings ??= new List<KeyMapping>();
            device.Mappings.RemoveAll(m => m.Key == keyName);
            device.Mappings.Add(new KeyMapping { Key = keyName, Action = action, Swallow = null });
        }

        _configStore.Save();
        MessageBox.Show("Saved.");
    }
}
```

```csharp
// src/Keysist/Ui/EditMappingsWindow.cs
using System.Windows.Forms;
using Keysist.Config;

namespace Keysist.Ui;

public class EditMappingsWindow : Form
{
    private readonly ConfigStore _configStore;
    private readonly TreeView _tree = new() { Dock = DockStyle.Fill };
    private readonly Button _deleteButton = new() { Dock = DockStyle.Bottom, Text = "Delete selected" };
    private readonly CheckBox _swallowBox = new() { Dock = DockStyle.Bottom, Text = "Swallow keystroke" };

    public EditMappingsWindow(ConfigStore configStore)
    {
        _configStore = configStore;
        Text = "Edit Mappings";
        Width = 500;
        Height = 400;

        Controls.Add(_tree);
        Controls.Add(_swallowBox);
        Controls.Add(_deleteButton);

        _deleteButton.Click += (_, _) => DeleteSelected();
        _swallowBox.CheckedChanged += (_, _) => ApplySwallowChange();
        _tree.AfterSelect += (_, _) => SyncSwallowCheckbox();

        Populate();
    }

    private void Populate()
    {
        _tree.Nodes.Clear();
        foreach (var device in _configStore.Current.Devices)
        {
            var deviceNode = new TreeNode($"{device.Name} ({device.DevicePathMatch})") { Tag = device };
            if (device.AnyKeyAction is not null)
            {
                deviceNode.Nodes.Add(new TreeNode($"[any key] -> {device.AnyKeyAction.Type}") { Tag = (device, (KeyMapping?)null) });
            }
            foreach (var mapping in device.Mappings ?? new List<KeyMapping>())
            {
                deviceNode.Nodes.Add(new TreeNode($"{mapping.Key} -> {mapping.Action.Type}") { Tag = (device, mapping) });
            }
            _tree.Nodes.Add(deviceNode);
        }
        _tree.ExpandAll();
    }

    private void SyncSwallowCheckbox()
    {
        if (_tree.SelectedNode?.Tag is (DeviceConfig device, KeyMapping? mapping))
        {
            _swallowBox.Enabled = true;
            _swallowBox.Checked = mapping?.Swallow ?? device.DefaultSwallow;
        }
        else
        {
            _swallowBox.Enabled = false;
        }
    }

    private void ApplySwallowChange()
    {
        if (_tree.SelectedNode?.Tag is (DeviceConfig _, KeyMapping mapping))
        {
            mapping.Swallow = _swallowBox.Checked;
            _configStore.Save();
        }
    }

    private void DeleteSelected()
    {
        if (_tree.SelectedNode?.Tag is DeviceConfig device)
        {
            _configStore.Current.Devices.Remove(device);
        }
        else if (_tree.SelectedNode?.Tag is (DeviceConfig owner, KeyMapping? mapping))
        {
            if (mapping is not null) owner.Mappings?.Remove(mapping);
            else owner.AnyKeyAction = null;
        }
        else
        {
            return;
        }

        _configStore.Save();
        Populate();
    }
}
```
